  \ifnum\getpagerefnumber{algorithm-start-#1}>0
    \ifnum\getpagerefnumber{algorithm-start-#1}>\value{page}\else

    \fi
  \fi
}

% Flush-bottom equivalent of needspace, taken from KACTL (CC0).
\newcommand{\BookNeedspace}[1]{%
  \begingroup
    \vskip#1\penalty9999\vskip-#1\vskip0pt
  \endgroup
}
\newcommand{\BookChapter}[1]{%
  \par\BookNeedspace{5\baselineskip}%
  \refstepcounter{section}%
  \pdfbookmark[0]{#1}{chapter-\thesection}%
  {\huge\bfseries\underline{#1} (\thesection)\par}\nobreak
  \vspace{3pt}%
}
\newcommand{\BookAlgorithm}[3]{%
  \par\BookNeedspace{4\baselineskip}\penalty-100
  \label{algorithm-start-#1}%
  \pdfbookmark[1]{#2}{algorithm-#1}%
  {\large #2\par}\nobreak
  {\tiny\hfill #3 lines\par}\nobreak
}
